# ENet reliable UDP networking
set(sources callbacks.c compress.c enet_dagor.c host.c list.c packet.c peer.c protocol.c)
if(WIN32)
  list(APPEND sources win32.c)
endif()
if(CMAKE_SYSTEM_NAME MATCHES "^(Android|Darwin|Linux|iOS|tvOS)$")
  list(APPEND sources unix.c)
endif()
set(includes ${DAGOR_PROG_DIR}/3rdPartyLibs/enet/include)
if(WIN32)
  list(APPEND libs System::winmm System::ws2_32)
elseif(CMAKE_SYSTEM_NAME MATCHES "^(Android|Darwin|Linux|iOS|tvOS)$")
  list(APPEND defines HAS_SOCKLEN_T HAS_POLL HAS_INET_NTOP HAS_INET_PTON HAS_MSGHDR_FLAGS SOMAXCONN=128)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "^(Clang|AppleClang|GNU)$")
  list(APPEND options -Wno-unused-parameter)
endif()

dagor_add_library(3rdPartyLibs.enet LAX ${sources})
target_include_directories(3rdPartyLibs.enet PRIVATE ${includes})
target_compile_definitions(3rdPartyLibs.enet PRIVATE ${defines})
target_compile_options(3rdPartyLibs.enet PRIVATE ${options})
target_link_libraries(3rdPartyLibs.enet PUBLIC ${libs})
