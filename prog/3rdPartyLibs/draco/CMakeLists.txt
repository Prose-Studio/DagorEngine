# Draco 1.5.7 (https://github.com/google/draco): the decoder of its compressed meshes and point clouds, for assimp's
# glTF 2 importer (KHR_draco_mesh_compression); its features are set in draco_features.h. The engine's build replaces
# upstream's CMakeLists.txt.
set(options)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  list(APPEND options -Wno-implicit-const-int-float-conversion) # attributes/geometry_attribute.h
endif()

dagor_add_target(NAME 3rdPartyLibs.draco STATIC LAX
  FILES_CMAKE draco_files.cmake
  INCLUDE_DIRECTORIES PRIVATE .
  COMPILE_DEFINITIONS PRIVATE
    DRACO_ENABLE_ENCODER=1 DRACO_ENABLE_DECODER=1 DRACO_MESH_COMPRESSION=1 DRACO_ANIMATION_COMPRESSION=1
    DRACO_ATTRIBUTE_COMPRESSION=1 DRACO_ATTRIBUTE_QUANTIZATION=1 DRACO_NORMAL_COMPRESSION=1
    DRACO_NORMAL_ENCODING_SUPPORTED=1 DRACO_TEXTURE_COORDINATE_COMPRESSION=1 DRACO_COLORS_COMPRESSION=1
    DRACO_JOINTS_WEIGHTS_COMPRESSION=1 DRACO_MESH_PREDICTION_SCHEMES=1 DRACO_POINT_CLOUD_PREDICTION_SCHEMES=1
    DRACO_STANDARD_EDGEBREAKER_SUPPORTED=1 DRACO_SEQUENTIAL_DECODING=1 DRACO_EDGEBREAKER_DECODING=1
    DRACO_KD_TREE_DECODING=1
  COMPILE_OPTIONS PRIVATE ${options}
)
