# the drivers of prog/engine/drv: the 3d drivers of the tree (DAGOR_D3D_DRIVERS), and the input drivers of the platform
add_subdirectory(drv3d_commonCode)
add_subdirectory(drv3d_stub)
add_subdirectory(shadersMetaData)
add_subdirectory(vr_device)
foreach(driver DX11 DX12 Metal vulkan)
  if(driver IN_LIST DAGOR_D3D_DRIVERS)
    add_subdirectory(drv3d_${driver})
  endif()
endforeach()
# the null driver: linked directly by the programs without a display, or one of the drivers of a multi-driver tree
if(NOT DAGOR_D3D_MULTI OR "null" IN_LIST DAGOR_D3D_DRIVERS)
  add_subdirectory(drv3d_null)
endif()
if(DAGOR_D3D_MULTI)
  add_subdirectory(drv3d_pc_multi)
endif()
